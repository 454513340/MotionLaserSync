TSIZE = 1600
CONST TBL_BASE   = 330
CONST POINT_DIM  = 3
CONST DATA_BASE  = TBL_BASE + 10
GLOBAL g_bufSize
GLOBAL g_wIndex
GLOBAL g_period
GLOBAL g_isrCount
GLOBAL g_triggerCount

SUB main()
    ' 管理区 330..339，数据区 340..1599：420槽；环形判满保留1槽。
    g_bufSize = (TSIZE - DATA_BASE) / POINT_DIM
    g_wIndex  = 0
    g_period  = 5          ' 名义定时周期ms；不等价于已测得的硬实时保证
    g_isrCount  = 0
    g_triggerCount = 0
    TABLE(TBL_BASE + 0) = g_bufSize
    TABLE(TBL_BASE + 1) = g_wIndex
    TABLE(TBL_BASE + 2) = 0
    TABLE(TBL_BASE + 3) = 0
    TABLE(TBL_BASE + 4) = 0
    TABLE(TBL_BASE + 5) = g_isrCount
    TABLE(TBL_BASE + 6) = g_triggerCount
    OP(0,OFF)
    INT_ENABLE = 1
    TIMER_START(0,g_period)
    WHILE 1
        DELAY(1000)
    WEND
END SUB

GLOBAL SUB ONTIMER0()
    ' 保留实机逻辑：每次回调先重启定时器。
    TIMER_START(0, g_period)
    DIM enableFlag
    DIM rIndex
    DIM nextIdx
    DIM baseAddr
    enableFlag = TABLE(TBL_BASE + 4)
    IF enableFlag <> 0 THEN
        OP(0, ON)          ' 开始本次输出脉冲命令，随后读取位置
        rIndex = TABLE(TBL_BASE + 2)
        nextIdx = g_wIndex + 1
        IF nextIdx >= g_bufSize THEN
            nextIdx = 0
        ENDIF
        IF nextIdx = rIndex THEN
            TABLE(TBL_BASE + 3) = 1   ' 锁存溢出；本BASIC继续执行，并非自动停触发
        ENDIF
        baseAddr = DATA_BASE + g_wIndex * POINT_DIM
        ' 顺序读取三轴指令位置，不是同一时刻锁存的编码器反馈。
        TABLE(baseAddr + 0) = DPOS(1)
        TABLE(baseAddr + 1) = DPOS(2)
        TABLE(baseAddr + 2) = DPOS(3)
        OP(0, OFF)         ' 结束输出脉冲命令，随后发布writeIndex
    ELSE
        ' 停用期间同步PC写回的触发计数；EXIT后不累计isrCount。
        g_triggerCount = TABLE(TBL_BASE + 6)
        EXIT SUB
    ENDIF
    g_wIndex = nextIdx
    TABLE(TBL_BASE + 1) = g_wIndex
    ' 仅统计采样使能期间完成的回调。PC每次启动不清零isrCount。
    g_isrCount =g_isrCount + 1
    TABLE(TBL_BASE+5) = g_isrCount
    g_triggerCount = g_triggerCount + 1
    TABLE(TBL_BASE + 6) = g_triggerCount
END SUB
